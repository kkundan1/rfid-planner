# Add with:  crontab -e   (as the user that runs the planner)
# Replace YOUR_CRON_SECRET with the value from .env.local

# Follow-up emails: checks every 15 minutes, sends once a day after the time set in Admin settings → Email
*/15 * * * * curl -s -X POST -H "x-cron-secret: YOUR_CRON_SECRET" http://127.0.0.1:3000/api/cron/followups > /dev/null

# Database backup every night at 01:30 (keeps the last 30)
30 1 * * * cd /var/www/rfid-planner && DATABASE_PATH=/var/lib/rfid-planner/planner.db BACKUP_DIR=/var/backups/rfid-planner /usr/bin/node scripts/backup.mjs >> /var/log/rfid-planner-backup.log 2>&1
